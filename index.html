// Automatically upload user's MP3 file to cloud storage in the background async function handleFileSelected(event) { const file = event.target.files[0]; if (!file) return; const toast = document.getElementById('toast'); showToast('Uploading song for WhatsApp sharing... ⏳'); try { const formData = new FormData(); formData.append('file', file); // Free background upload API (no account needed, CORS friendly) const response = await fetch('https://tmpfiles.org/api/v1/upload', { method: 'POST', body: formData }); const data = await response.json(); if (data.status === 'success') { // Convert view URL to direct streaming MP3 URL const directMp3Url = data.data.url.replace('tmpfiles.org/', 'tmpfiles.org/dl/'); customAudioUrl = directMp3Url; currentMusicSource = 'url'; // Update UI & Local storage document.getElementById('inputMusicUrl').value = directMp3Url; localStorage.setItem('saved_birthday_music', directMp3Url); customAudioPlayer.src = directMp3Url; showToast('Song ready to share! 🎵🎉'); } else { throw new Error('Upload failed'); } } catch (err) { console.error('Auto upload error:', err); // Fallback to local playback if offline or error customAudioUrl = URL.createObjectURL(file); customAudioPlayer.src = customAudioUrl; currentMusicSource = 'localFile'; showToast('Loaded locally (sharing may limited) ⚠️'); } }
